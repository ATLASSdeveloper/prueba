import { createHash } from "crypto";
import { Request, Response, Router } from "express";
import { Pool } from "pg";
import { z } from "zod";

const pool = new Pool();
const router = Router();

interface AuthenticatedRequest extends Request {
  auth?: {
    userId: string;
    clienteId: string;
  };
}

const eventSchema = z
  .object({
    tipo: z.string().trim().min(1).max(100),
    descripcion: z.string().trim().max(1000).optional(),
    fecha: z.string().datetime().optional(),
  })
  .strict();

const idempotencyKeySchema = z.string().trim().min(1).max(255);

export async function createEvent(
  req: AuthenticatedRequest,
  res: Response,
): Promise<void> {
  // (1) autorización en servidor: el usuario autenticado (asume req.auth = { userId, clienteId }) solo puede escribir en SU clienteId; rechaza el resto con 403.
  const auth = req.auth;
  if (!auth?.userId || !auth.clienteId) {
    res.status(401).json({ error: "Unauthenticated" });
    return;
  }
  if (auth.clienteId !== req.params.clienteId) {
    res.status(403).json({ error: "Forbidden" });
    return;
  }

  // (4) Validación de entrada del body.
  const eventResult = eventSchema.safeParse(req.body);
  const keyResult = idempotencyKeySchema.safeParse(
    req.get("Idempotency-Key"),
  );
  if (!eventResult.success || !keyResult.success) {
    res.status(400).json({ error: "Invalid request" });
    return;
  }

  const event = eventResult.data;
  const idempotencyKey = keyResult.data;
  const payloadHash = createHash("sha256")
    .update(JSON.stringify(event))
    .digest("hex");

  try {
    // (2) Consulta a base parametrizada (nada de concatenar strings SQL).
    // (3) Idempotencia: acepta una cabecera Idempotency-Key y evita insertar duplicados.
    const insertResult = await pool.query(
      `INSERT INTO eventos
         (cliente_id, tipo, descripcion, fecha, idempotency_key, payload_hash)
       VALUES ($1, $2, $3, $4, $5, $6)
       ON CONFLICT (cliente_id, idempotency_key) DO NOTHING
       RETURNING id`,
      [
         auth.clienteId,
         event.tipo,
         event.descripcion ?? null,
         event.fecha ?? null,
        idempotencyKey,
        payloadHash,
      ],
    );

    if (insertResult.rowCount === 1) {
      res.status(201).json({ id: insertResult.rows[0].id });
      return;
    }

    const existingResult = await pool.query(
      `SELECT id, payload_hash
       FROM eventos
       WHERE cliente_id = $1 AND idempotency_key = $2`,
      [auth.clienteId, idempotencyKey],
    );
    const existingEvent = existingResult.rows[0];
    if (!existingEvent) {
      throw new Error("Idempotency conflict without a corresponding event");
    }
    if (existingEvent.payload_hash !== payloadHash) {
      res.status(409).json({
        error: "Idempotency key was already used with a different payload",
      });
      return;
    }

    res.status(200).json({ id: existingEvent.id });
  } catch (error) {
    // (5) Manejo de errores con códigos HTTP correctos y sin filtrar detalles internos al cliente
    console.error("Failed to insert event", error);
    res.status(500).json({ error: "Internal server error" });
  }
}

router.post("/clientes/:clienteId/eventos", createEvent);

export default router;
